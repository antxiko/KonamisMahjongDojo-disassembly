# =========================================================================
# LA ZONA LIBRE
# =========================================================================
# Los 8.219 bytes de 0xFF del final del cartucho (0x9FE5-0xBFFF). Aqui van
# los bloques que en ingles ocupan mas que en japones: el ingles necesita mas
# celdas que el katakana para decir lo mismo.
#
# OJO con lo que esto promete: que el chip original tuviera esos 8 KB
# grabables es una SUPOSICION (ver .forja/mission.md). En emulador y en
# flashcart -que es como se juega- funciona seguro.

[libre DATA_relleno_del_final org=0x9FE5 fin=0xC000]

# EL LOGO DEL TITULO
# =========================================================================
# El cartucho pone ahi 麻雀道場 (Mahjong Dojo) en cuatro kanji de 32x24, con
# los tiles 0xA0-0xD5 del tercio 0: dieciocho columnas por tres filas, o sea
# un lienzo de 144x24 pixeles. Aqui va el nombre en letras latinas, al estilo
# de los logos de Konami de esos anos.
#
# Los tiles van ordenados POR COLUMNAS -A0 es f4c7, A1 f5c7, A2 f6c7 y A3
# vuelve arriba en la c8-, que no es como reparte los glifos grandes la

# =========================================================================
# PASO 6: EL MODO ATTRACT
# =========================================================================
# Un pase de diapositivas que ensena a jugar, escrito en los bytes libres del
# final del cartucho. NO usa el motor del juego: solo pinta y pasa de pantalla.
#
# DONDE SE ENGANCHA, y por que ahi: el estado 14 es la cuenta atras con el
# rotulo END puesto, y al agotarse salta a `vuelve_al_estado_0` desde 0x43FD.
# Es el UNICO salto del cartucho a esa rutina y justo el punto por donde el
# demo cierra su ciclo, asi que el attract se cuela entre el final de una
# vuelta y el principio de la siguiente, que es donde va un attract.
#
# EL RELOJ SALE GRATIS. El estado 14 hace `dec (0xE004)` uno de cada cuatro
# fotogramas y llama a 0x43FD al llegar a cero. Recargando 0xE004 y haciendo
# `ret`, el juego SIGUE en el estado 14 y vuelve a llamarnos: el temporizador
# de las diapositivas es el que ya tenia el cartucho. Cero codigo de espera.
#
# EL CONTADOR DE DIAPOSITIVA ES 0xE001, el submodo, y tampoco cuesta nada:
#   - al entrar en el estado 14 vale 0 SIEMPRE, porque el 13 sale por
#     `avanza_de_estado` (0x41FF), que hace `ld (0xE001),0`. MEDIDO en el
#     emulador: los volcados del final salen todos como e14_s00.
#   - el estado 14 no lo toca en ningun momento;
#   - y las dos salidas posibles lo dejan a cero solas: `vuelve_al_estado_0`
#     al terminar el pase, y `arranca_la_partida` (0x4ABC) si alguien pulsa
#     1, 2 o 3, que escribe 8 y 0 de una tacada en 0xE000/0xE001.
# Por eso el attract se abandona limpiamente en cuanto el jugador pulsa: la
# puerta del teclado sigue viva porque el estado 14 ya ha apagado el bit 6 de
# 0xE002 antes de llamarnos.
#
# CON QUE SE PINTA, medido sobre el volcado del estado 14 (work/vram_attract,
# tools/dibuja_las_fichas.py y tools/pantalla.py --hoja) y no supuesto:
#   - los tres tercios llegan IDENTICOS byte a byte, patrones y colores, que
#     los iguala `replica_patrones_y_colores_en_los_tercios`. Se puede pintar
#     en las 24 filas con el mismo juego de tiles, y una hilera de fichas
#     puede cruzar de un tercio al siguiente sin partirse.
#   - el alfabeto latino del parche esta en 0x30-0x49 (charset `recuento`),
#     blanco sobre negro, y el tile 0x01 es el negro.
#   - de las 34 fichas solo llegan enteras las nueve de CARACTERES, las nueve
#     de BAMBUES, los circulos 1 y 2, el dorso y el hueco. FALTAN los siete
#     honores y los circulos del 3 al 9: dieciseis fichas de treinta y cuatro.
#   - los DIGITOS no valen aqui: 0x10-0x19 son los trozos del rotulo END.
#     Por eso las diapositivas no llevan ni una cifra.
#
# LAS DIECISEIS FICHAS QUE FALTAN, Y COMO SE METEN. Un tutorial de mahjong que
# nombra los vientos y los dragones sin ensenarlos no ensena nada, asi que hay
# que cargarlos. El sitio de siempre no sirve -los honores viven en los tiles
# 0x2E-0x57 y ahi esta el alfabeto del parche; los circulos 3-9 en 0x58-0x81 y
# ahi no hay patron ninguno-, pero los numeros de tile no tienen nada de
# sagrado: el dibujo se puede cargar en CUALQUIER hueco y pedirlo por su numero
# nuevo. Contando los tiles que el attract no dibuja jamas quedan libres
# 0x02-0x2D (el rotulo END, que ya no se vuelve a pintar) y 0x4E-0x81, y con
# eso caben las dieciseis:
#     honores 0x2E-0x57 -> 0x4E-0x77      circulos 3-9 0x58-0x81 -> 0x02-0x2B
#     dorso   0x28-0x2D -> 0x78-0x7D      (lo echan los circulos de su sitio)
# Los patrones y los colores salen de los bloques de la ROM ORIGINAL (0x9343 y
# 0x99F9, el par del tercio de abajo), NO del bloque ya parcheado: el charset
# `cajas` mete letras en los tiles 0x49, 0x4F, 0x54 y 0x55, que caen dentro del
# rango de los honores.
#
# Y SE CARGAN EN UN SOLO TERCIO. Escribir los tres cuesta el triple; como la
# pantalla del final trae los tres identicos -medido-, basta con dejarlos en el
# primero y llamar a `replica_patrones_y_colores_en_los_tercios`, la misma
# rutina que usa el cartucho en 0x43B3, tambien desde la interrupcion.
#
# LA LETRA A HAY QUE CARGARLA, y esto costo una medida: en esta pantalla el
# tile 0x30 no es la A sino el VERDE del interior de la caja del final
# (descomprimiendo 0x87B4 salen dieciocho 0x30 por fila), asi que el bloque de
# patrones del rotulo END lo restaura desde la ROM y se lleva la A por delante.
# En el cartucho original daba igual; aqui la primera prueba escribia
# "SECOND ONE / AND BACK" con dos cuadrados verdes donde iban las aes.
# No se puede arreglar en el bloque del final -ese verde SI se ve, durante los
# 64 fotogramas que tarda el muro de fichas en taparlo-, asi que el attract se
# carga su propia A antes de la primera diapositiva.

[asmlibre attract]
; ----------------------------------------------------------------------
; EL MODO ATTRACT. Entra desde el `jp` de 0x43FD, una vez por diapositiva.
; Devuelve el mando al estado 14, que es quien cuenta, hasta que se acaban.
; ----------------------------------------------------------------------
attract:
	ld hl,0e001h		; el submodo, que aqui cuenta diapositivas
	ld a,(hl)
	cp attract_cuantas
	jr nc,attract_termina	; vistas todas: el ciclo del demo sigue su camino
	inc (hl)
	push af
	call attract_limpia	; la pantalla, a negro: viene con el rotulo END
	pop af
	or a			; en la primera hay que cargar los tiles que faltan
	jr nz,attract_pinta
	push af
	call attract_carga_los_tiles	; y se cargan con la pantalla YA en negro:
	pop af				; tardan 1,24 s medidos, y ese rato se veria
attract_pinta:				; el rotulo END con los tiles ya pisados
	add a,a			; dos bytes por entrada
	ld hl,attract_tabla
	call suma_a_a_hl
	ld e,(hl)
	inc hl
	ld d,(hl)
	ex de,hl
	call L_409D		; el interprete de formato A, por la puerta que PINTA
	ld a,050h
	ld (0e004h),a		; 80 x 4 fotogramas: 6,4 s, que dan para leer la pantalla
	ret			; y a seguir en el estado 14, que volvera a llamarnos
attract_termina:
	jp vuelve_al_estado_0

; ----------------------------------------------------------------------
; Las dieciseis fichas que a esta pantalla le faltan, mas la letra A. Todo al
; PRIMER tercio, y luego la rutina del cartucho lo reparte a los otros dos: la
; pantalla del final llega con los tres identicos, asi que replicar el tercio
; entero no pisa nada y sale por la tercera parte de los bytes.
; ----------------------------------------------------------------------
attract_carga_los_tiles:
	ld hl,attract_los_honores	; vientos y dragones, a los tiles 0x4E-0x77
	call pinta_lista_formato_b
	ld hl,attract_los_circulos	; los circulos 3-9, a 0x02-0x2B
	call pinta_lista_formato_b
	ld hl,attract_el_dorso		; y el dorso, que se queda sin su sitio
	call pinta_lista_formato_b
	ld hl,attract_la_letra_a
	call L_409D
	jp replica_patrones_y_colores_en_los_tercios

; ----------------------------------------------------------------------
; Pone toda la tabla de nombres al tile negro. Con formato A costaria los 768
; bytes de la lista; asi son quince. El puerto de datos lo deja el propio
; prepara_escritura_vram en C', y `exx` dos veces devuelve B y C intactos, que
; es como vuelca el cartucho en 0x46C0.
; ----------------------------------------------------------------------
attract_limpia:
	ld de,03800h
	call prepara_escritura_vram
	ld c,003h		; tres pasadas de 256 = las 768 celdas
	ld b,000h
attract_borra:
	ld a,001h		; el tile negro de esta pantalla
	exx
	out (c),a
	exx
	djnz attract_borra
	dec c
	jr nz,attract_borra
	ei
	ret

attract_tabla:
	defw attract_1
	defw attract_2
	defw attract_3
	defw attract_4
	defw attract_5
	defw attract_6
	defw attract_7
	defw attract_8
	defw attract_9
	defw attract_10
	defw attract_11
	defw attract_12
	defw attract_13
	defw attract_14
attract_cuantas: equ 14


# Los tiles que faltan, sacados de la ROM ORIGINAL y reubicados. A partir de
# aqui las diapositivas piden HON1..HON7, PIN3..PIN9 y DORSO por su nombre de
# siempre y salen los numeros nuevos: los reescribe el generador.

[tilesnuevos attract_los_honores patrones=0x9343 colores=0x99F9 a=0x4E]

[tilesnuevos attract_el_dorso patrones=0x9343 colores=0x99F9 a=0x78]

# que de repartirlo ya se encarga la rutina del cartucho. Es una lista de
