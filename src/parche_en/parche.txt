# =========================================================================
# LA ZONA LIBRE
# =========================================================================
# Los 8.219 bytes de 0xFF del final del cartucho (0x9FE5-0xBFFF). Aqui van
# los bloques que en ingles ocupan mas que en japones: el ingles necesita mas
# celdas que el katakana para decir lo mismo.
#
# OJO con lo que esto promete: que el chip original tuviera esos 8 KB
# grabables es una SUPOSICION (ver .forja/mission.md). En emulador y en
# flashcart -que es como se juega- funciona seguro.

[libre DATA_relleno_del_final org=0x9FE5 fin=0xC000]

# EL LOGO DEL TITULO
# =========================================================================
# El cartucho pone ahi 麻雀道場 (Mahjong Dojo) en cuatro kanji de 32x24, con
# los tiles 0xA0-0xD5 del tercio 0: dieciocho columnas por tres filas, o sea
# un lienzo de 144x24 pixeles. Aqui va el nombre en letras latinas, al estilo
# de los logos de Konami de esos anos.
#
# Los tiles van ordenados POR COLUMNAS -A0 es f4c7, A1 f5c7, A2 f6c7 y A3
# vuelve arriba en la c8-, que no es como reparte los glifos grandes la

# =========================================================================
# FASE 3: LOS ROTULOS DE LA ESPERA
# =========================================================================
# Debajo del recuento, el cartucho dice con que espera se gano la mano y si
# fue por tsumo o por ron. Es lo ultimo que quedaba en katakana, y ahora
# canta porque todo lo de arriba esta en ingles.
#
# Aqui tampoco caben letras de 8x8 -RYANMEN son siete y el hueco seis
# celdas-, asi que va con la misma fuente estrecha de dos letras por tile.
# Las cuentas cuadran JUSTO: 17 digrafos distintos y 17 tiles libres, y el
# 0x27 se queda para FU, que en digrafos es un tile solo igual que antes.
# El reparto lo hace tools/reparte_digrafos_espera.py; se regenera.

# Los cuatro que salen del bloque SIN comprimir.
[crudos DATA_patrones_sin_comprimir_3020 base=rom destino=0x7020 tiles=0x04-0x0F]
08 estrecho "AN"
09 estrecho "FU"
0A estrecho "HA"
0B estrecho "I "
0C estrecho "KA"
0D estrecho "ME"
0E estrecho "N "
0F estrecho "NC"

# Y los trece del bloque grande, que se lee DOS veces: entero desde su
# principio y otra vez desde 0x93AA para las manos dibujadas del tablero.
# Esa segunda entrada cae justo en el tile 0x28, asi que los de antes se
# pueden cambiar recomprimiendo solo esa parte y reapuntando la instruccion.

[patrones DATA_patrones_del_tercio_de_abajo base=rom destino=0x70D0 tiles=0x1A-0xFF cola=0x93AA reapunta=pinta_las_dos_manos_del_tablero]
1A estrecho "NK"
1B estrecho "O "
1C estrecho "PE"
1D estrecho "PO"
1E estrecho "RO"
1F estrecho "RY"
20 estrecho "SH"
21 estrecho "TA"
22 estrecho "TS"
23 estrecho "UM"
# Las siete tiras. El tsumo empieza en 0x744A y no en 0x744B: SHANPON baja
# de seis celdas a cuatro y deja ese byte libre, que es justo el que le
# faltaba a TSUMO (tres tiles y el remate).

[cuerpos DATA_rotulos_de_la_espera charset=digrafos]
0x7428 "RYANMEN"
0x742F "KANCHAN"
0x7436 "PENCHAN"
0x743D "TANKI"
0x7444 "SHANPON"
0x744A "TSUMO"
0x744E "RON"

[codigo L_72E4]
ld hl,0744bh => ld hl,0744ah

# Y los marcos de los fu: sueltan su borde izquierdo (el tile 0x26, que hace
# falta para el reparto) y su フ pasa a decir FU, que en digrafos es un
# tile solo igual que antes. Ese FU va al bloque SIN comprimir porque el
# comprimido no daba para un tile mas.

[formatoA DATA_marcos_de_los_fu charset=digrafos]
f16 c10  {01 01 01 09}
f16 c22  {01 01 01 09}
f19 c10  {01 01 01 09}
f19 c22  {01 01 01 09}
f22 c10  {01 01 01 09}
f22 c22  {01 01 01 09}


# =========================================================================
# EL ROTULO DEL FINAL DE LA PARTIDA
# =========================================================================
# Al acabar la partida el cartucho pone 終局 (shuukyoku, fin de la partida) en
# dos kanji de 24x24, en las filas 10 a 12 y las columnas 12-14 y 17-19. Son
# DIECIOCHO tiles (0x10-0x21), que es justo lo que ocupan tres letras de dos
# tiles de ancho por tres de alto: END entra exacto, y sobra sitio a los lados
# para centrarlo en las diez celdas del rotulo.

[patrones DATA_patrones_del_final_080 base=rom destino=0x6080 tiles=0x10-0x30]
10-15 E@final
16-1B N@final
1C-21 D@final

# Las tres filas del rotulo, con las diez celdas escritas enteras para que no
# quede nada del kanji a los lados, y una celda en blanco entre letra y letra:
# 1+2+1+2+1+2+1 = 10 clavadas.

[formatoA DATA_logotipo_de_tres_filas]
f10 c11  {01 10 11 01 16 17 01 1C 1D 01}
f11 c11  {01 12 13 01 18 19 01 1E 1F 01}
f12 c11  {01 14 15 01 1A 1B 01 20 21 01}
